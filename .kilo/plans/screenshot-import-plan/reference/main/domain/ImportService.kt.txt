package com.example.expenses.importer.domain

import com.example.expenses.importer.dedupe.DuplicateDetector
import com.example.expenses.importer.dedupe.DuplicateStatus
import com.example.expenses.importer.model.Bank
import com.example.expenses.importer.model.OcrResult
import com.example.expenses.importer.parsing.BankStatementParser
import com.example.expenses.importer.parsing.ParserRegistry
import java.time.LocalDate
import kotlin.coroutines.cancellation.CancellationException

class ImportService(
    private val existing: ExistingTransactionsSource,
    private val categories: VendorCategoryLookup,
    private val normaliser: VendorNameNormalizer,
    private val writer: TransactionWriter,
    private val parserFor: (Bank) -> BankStatementParser = ParserRegistry::forBank,
) {
    suspend fun prepare(ocr: OcrResult, bank: Bank, today: LocalDate): PreparedImport {
        val parsed = parserFor(bank).parse(ocr, today)
        val warnings = parsed.warnings.toMutableList()
        if (parsed.transactions.isEmpty()) return PreparedImport(emptyList(), warnings)

        val from = parsed.transactions.minOf { it.date }.minusDays(1)
        val to = parsed.transactions.maxOf { it.date }.plusDays(1)
        val saved = try {
            existing.between(from, to)
        } catch (e: CancellationException) {
            throw e
        } catch (e: Exception) {
            warnings += "Could not check for duplicates"
            emptyList()
        }
        val statuses = DuplicateDetector.detect(parsed.transactions, saved)

        val items = parsed.transactions.mapIndexed { index, p ->
            val vendor = normaliser.normalise(p.vendor).ifBlank { p.vendor }
            ReviewItem(
                id = index,
                selected = !p.pending && statuses[index] != DuplicateStatus.LIKELY,
                date = p.date,
                vendor = vendor,
                amountMinor = p.amountMinor,
                kind = p.kind,
                category = categories.categoryFor(vendor, p.kind),
                duplicate = statuses[index],
                pending = p.pending,
                dateUncertain = p.dateUncertain,
                rawText = p.rawText,
            )
        }
        return PreparedImport(items, warnings)
    }

    suspend fun commit(items: List<ReviewItem>): CommitResult {
        var saved = 0
        var failed = 0
        var skippedNoCategory = 0
        for (item in items.filter { it.selected }.sortedBy { it.date }) {
            val category = item.category
            if (category == null) {
                skippedNoCategory++
                continue
            }
            try {
                writer.add(NewTransaction(item.date, item.vendor, item.amountMinor, item.kind, category.id))
                saved++
            } catch (e: CancellationException) {
                throw e
            } catch (e: Exception) {
                failed++
            }
        }
        return CommitResult(saved, failed, skippedNoCategory)
    }
}
