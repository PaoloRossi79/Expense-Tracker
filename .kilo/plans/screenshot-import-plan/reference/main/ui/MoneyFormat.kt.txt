package com.example.expenses.importer.ui

import com.example.expenses.importer.model.TransactionKind
import java.math.BigDecimal
import java.text.NumberFormat
import java.util.Locale

/** 1250 EXPENSE -> "£12.50", 250000 INCOME -> "+£2,500.00". */
fun formatMinor(amountMinor: Long, kind: TransactionKind): String {
    val format = NumberFormat.getNumberInstance(Locale.UK).apply {
        minimumFractionDigits = 2
        maximumFractionDigits = 2
    }
    val body = "£" + format.format(BigDecimal(amountMinor).movePointLeft(2))
    return if (kind == TransactionKind.INCOME) "+$body" else body
}

private val COMMA_DECIMAL = Regex("^\\d+,\\d{1,2}$")            // 12,5   12,50
private val COMMA_THOUSANDS = Regex("^\\d{1,3}(,\\d{3})+(\\.\\d{1,2})?$")   // 1,234   1,234.50
private val PLAIN = Regex("^\\d+(\\.\\d{1,2})?$")                // 12   12.5   12.50

/** "12.5" -> 1250, "12,50" -> 1250, "1,234.50" -> 123450, anything else (negative, 3 decimals, text) -> null. */
fun parseMinor(text: String): Long? {
    val t = text.trim().removePrefix("£").trim()
    val normalised = when {
        COMMA_DECIMAL.matches(t) -> t.replace(',', '.')
        COMMA_THOUSANDS.matches(t) -> t.replace(",", "")
        PLAIN.matches(t) -> t
        else -> return null
    }
    return try {
        BigDecimal(normalised).movePointRight(2).longValueExact()
    } catch (e: ArithmeticException) {
        null
    }
}
